▏ 每个玩家的 playerComp 需要唯一的 IP 和位置，不能所有客户端都用同一个种子生成的同一个值。

    int playerSlot = GetPlayerSlot(); // 0 = 服务端管理员, 1 = 客户端A, 2 = 客户端B

    // 用 slot 派生的种子重新生成 IP
    string playerIP = GenerateUniqueIP(masterSeed, playerSlot);
    // 在 new OS() 之后立即覆盖 thisComputer.ip
▏ 
▏ 更具体的实现：
▏ 
    // 在 new OS() 之前，先注入一个派生子种子
    // 这样 playerComp 的 IP 会基于 (masterSeed + playerSlot) 生成
    int childSeed = WorldSeedSync.GetNextChildSeed(playerSlot);
    Utils.random.ReSeed(childSeed);
    var oS = new OS();
    oS.thisComputer.ip = oS.thisComputer.adminIP; // 确保 adminIP 和 IP 一致

▏

▏ idName 是 "playerComp"，并且源 IP 不是自己的 IP，就忽略这条事件。
▏ 
    收到 EVENT:NODE_IP:playerComp:192.168.1.1
